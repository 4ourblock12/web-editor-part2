/** * Proxy CurseForge untuk Title Maker. * * Setup (gratis): * 1. Daftar di https://dash.cloudflare.com -> Workers & Pages -> Create -> Worker. * 2. Tempel kode ini lalu Deploy. * 3. Settings -> Variables and Secrets: * CF_API_KEY (Secret) = API key dari https://console.curseforge.com * ALLOWED_ORIGIN (Text) = https://4ourblock12.github.io * 4. Salin URL worker (https://nama.akun.workers.dev) ke kolom "URL proxy CurseForge". * * Endpoint: * GET /v1/... -> diteruskan ke https://api.curseforge.com/v1/... * GET /image?url=... -> mengambil gambar dari CDN yang diizinkan (agar canvas tidak kena CORS) */ const IMAGE_HOSTS = ['media.forgecdn.net', 'cdn.modrinth.com', 'avatars.githubusercontent.com']; export default { async fetch(request, env) { const origin = env.ALLOWED_ORIGIN || '*'; const cors = { 'Access-Control-Allow-Origin': origin, 'Access-Control-Allow-Methods': 'GET, OPTIONS', 'Access-Control-Allow-Headers': 'Accept, Content-Type', 'Vary': 'Origin' }; if (request.method === 'OPTIONS') return new Response(null, { status: 204, headers: cors }); if (request.method !== 'GET') return new Response('Method not allowed', { status: 405, headers: cors }); // Batasi hanya untuk origin web Anda const reqOrigin = request.headers.get('Origin'); if (env.ALLOWED_ORIGIN && reqOrigin && reqOrigin !== env.ALLOWED_ORIGIN) { return new Response('Forbidden', { status: 403, headers: cors }); } const url = new URL(request.url); // ---- Proxy gambar ---- if (url.pathname === '/image') { let target; try { target = new URL(url.searchParams.get('url')); } catch { return new Response('Bad url', { status: 400, headers: cors }); } if (target.protocol !== 'https:' || !IMAGE_HOSTS.includes(target.hostname)) { return new Response('Host not allowed', { status: 403, headers: cors }); } const img = await fetch(target.toString(), { cf: { cacheTtl: 86400, cacheEverything: true } }); return new Response(img.body, { status: img.status, headers: { ...cors, 'Content-Type': img.headers.get('Content-Type') || 'image/png', 'Cache-Control': 'public, max-age=86400' } }); } // ---- Proxy API CurseForge ---- if (url.pathname.startsWith('/v1/')) { const api = await fetch('https://api.curseforge.com' + url.pathname + url.search, { headers: { 'x-api-key': env.CF_API_KEY, 'Accept': 'application/json' }, cf: { cacheTtl: 300, cacheEverything: true } }); return new Response(api.body, { status: api.status, headers: { ...cors, 'Content-Type': 'application/json', 'Cache-Control': 'public, max-age=300' } }); } return new Response('Not found', { status: 404, headers: cors }); } };
